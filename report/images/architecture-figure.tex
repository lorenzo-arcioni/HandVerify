% Layer 2
\node[layer, below=of fc1, fill=white] (fc2) {FC Layer 2 \hfill {\scriptsize → 256}};
\node[operation, right=0.3cm of fc2.east, anchor=west] (op2) {Linear\\BatchNorm\\ReLU\\Dropout(0.5)};

% Layer 3
\node[layer, below=of fc2, fill=white] (fc3) {FC Layer 3 \hfill {\scriptsize → 128}};
\node[operation, right=0.3cm of fc3.east, anchor=west] (op3) {Linear\\BatchNorm\\ReLU\\Dropout(0.4)};

% Layer 4
\node[layer, below=of fc3, fill=white] (fc4) {FC Layer 4 \hfill {\scriptsize → 64}};
\node[operation, right=0.3cm of fc4.east, anchor=west] (op4) {Linear\\BatchNorm\\ReLU\\Dropout(0.3)};

% Layer 5
\node[layer, below=of fc4, fill=white] (fc5) {FC Layer 5 \hfill {\scriptsize → 32}};
\node[operation, right=0.3cm of fc5.east, anchor=west] (op5) {Linear\\BatchNorm\\ReLU\\Dropout(0.3)};

\node[layer, below=of fc5, fill=green!15] (output_layer) {Output Layer \hfill {\scriptsize → 1}};

% Arrows

}

% Output
\node[layer, below=of output_layer, fill=black!10] (final) {Similarity Score: $s \in [0,1]$};
\draw[-Stealth, thick] (output_layer) -- (final);

% Title
\node[above=0.3cm of input, font=\small\bfseries] {Projection Head Architecture (BCE Loss)};

% Equation annotation
\node[right=1.5cm of fc3, align=left, font=\scriptsize] (eq) {
    $\mathbf{h}^{(l+1)} = \text{Dropout}($\\

    \hspace{1cm}$\text{BN}(\mathbf{W}^{(l)}\mathbf{h}^{(l)} + \mathbf{b}^{(l)})))$
};

% Caption
\node[below=0.3cm of final, align=center, font=\small] {
    \textbf{(b)} Projection Head for Binary Cross-Entropy Loss
};
