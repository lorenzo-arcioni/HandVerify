\title[HandVerify: Handwriting Verification]{HandVerify: A Comparative Study of Deep Learning Architectures for Writer Verification}

\author*[1]{\fnm{First} \sur{Author}}\email{first.author@university.edu}
\equalcont{These authors contributed equally to this work.}

\author[1]{\fnm{Second} \sur{Author}}\email{second.author@university.edu}
\equalcont{These authors contributed equally to this work.}

\affil*[1]{\orgdiv{Department of Computer Science}, \orgname{Sapienza University of Rome}, \orgaddress{\street{Via Salaria 113}, \city{Rome}, \postcode{00198}, \state{Italy}}}

\abstract{Writer verification is a critical biometric task with applications in forensic document analysis, security systems, and authentication. This paper presents HandVerify, a comprehensive framework for offline handwriting verification using deep Siamese neural networks. We systematically evaluate three learning paradigms: Binary Cross-Entropy (BCE) classification, Contrastive Learning, and Triplet Loss, across ten state-of-the-art CNN architectures including ResNet, EfficientNet and MobileNet families. We conduct extensive experiments on two benchmark datasets: RIMES (French) and IAM (English), enabling evaluation across different languages. Through rigorous 5-fold cross-validation, we demonstrate that appropriate architecture selection combined with task-specific loss functions achieves Equal Error Rates (EER) below 5\% on both datasets. Our results reveal that metric learning approaches (Contrastive and Triplet losses) consistently outperform BCE classification, particularly when combined with lightweight architectures such as MobileNetV3. The modular design of HandVerify enables rapid prototyping and fair comparison across architectural choices, establishing strong baselines for future research. All code and experimental protocols are publicly available.}

\keywords{Handwriting Verification, Siamese Networks, Contrastive Learning, Triplet Loss, Computer Vision, Deep Learning, Biometric Authentication, Transfer Learning}
